# DK-1624: Проверена переносимость чатов между Claude и Codex

## Что происходит

В чате можно менять модель и подписку с сохранением разговора. Текущий перенос использует совместимые транскрипты Claude Code. У Codex есть профиль для разовых поручений, но работа с ним через app-server ещё не подключена. Не проверено, можно ли передать историю из Claude в Codex и вернуть результат обратно без потери доступного пользователю контекста. Предмет согласован в разговоре 2026-11-06.

python3 - <<'PYCODE'
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import tempfile
import uuid

script = Path("tools/chat-port-check/native_projection.py").resolve()
with tempfile.TemporaryDirectory(prefix="dk1624-accept-") as temporary:
    root = Path(temporary)
    seed_id = str(uuid.uuid4())
    seed = root / (seed_id + ".jsonl")
    history = root / "history.jsonl"
    def save(path, rows):
        path.write_text("".join(json.dumps(r) + "\n" for r in rows))
    save(seed, [
        {"type": "user", "sessionId": seed_id, "cwd": str(root),
         "version": "2.1.312", "message": {"content":
         "Reply only RESUME_PROBE_READY.\nSERVICE_DO_NOT_COPY"}},
        {"type": "assistant", "sessionId": seed_id, "cwd": str(root),
         "message": {"content": [{"type": "thinking", "thinking": "FAKE"}]}},
        {"type": "assistant", "sessionId": seed_id, "cwd": str(root),
         "message": {"content": [{"type": "text", "text": "RESUME_PROBE_READY"}]}}
    ])
    call = {"type": "tool_use", "id": "call-accept", "name": "Bash",
            "input": {"command": "printf SYNTHETIC_ONLY"}}
    result = {"type": "tool_result", "tool_use_id": "call-accept",
              "content": "SYNTHETIC_ONLY", "is_error": True}
    save(history, [
        {"type": "assistant", "message": {"content": [call]}},
        {"type": "user", "message": {"content": [result]}}
    ])
    before = seed.read_bytes()
    command = [sys.executable, str(script), "--seed", str(seed), "--history",
               str(history), "--seed-uuid", seed_id, "--cwd", str(root),
               "--model", "synthetic-receiver"]
    completed = subprocess.run(command, capture_output=True, text=True)
    assert completed.returncode == 0, completed.stderr
    imported = next(line.split(": ", 1)[1] for line in completed.stdout.splitlines()
                    if line.startswith("native uuid: "))
    assert str(uuid.UUID(imported)) == imported and imported != seed_id
    target = root / (imported + ".jsonl")
    payload = target.read_bytes()
    rows = [json.loads(line) for line in payload.splitlines()]
    assert len(rows) == 4 and len({r["uuid"] for r in rows}) == 4
    assert all(r["sessionId"] == imported for r in rows)
    assert rows[0]["parentUuid"] is None
    assert all(b["parentUuid"] == a["uuid"] for a, b in zip(rows, rows[1:]))
    assert rows[2]["message"]["content"] == [call]
    assert rows[3]["message"]["content"] == [result]
    assert b"SERVICE_DO_NOT_COPY" not in payload and b"FAKE" not in payload
    assert seed.read_bytes() == before
    spec = importlib.util.spec_from_file_location("projection_accept", script)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    try:
        module.write_journal(seed, imported, rows)
    except module.ProjectionError:
        pass
    else:
        raise AssertionError("existing journal was accepted")
    assert target.read_bytes() == payload
