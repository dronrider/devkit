## Черновик

Утилита замера расхода контекста по транскриптам сессий. Разовый скрипт свёл ~/.claude/projects (972 сессии, 665 МБ, 57861 вызов инструментов) за пять секунд и показал, что интуиция промахивается: grep самая частая команда Bash (34.4% вызовов) и одна из самых дешёвых (1.2K символов в среднем), sed в 79.9% случаев не правка файла, а чтение фрагмента через -n Np (правок -i всего 175 за всю выборку), head и tail почти всегда стоят ограничителем в хвосте конвейера, а половину всего вывода инструментов даёт Read. Вызовы тяжелее 20K символов это 1.33% вызовов и 23.4% объёма, то есть резать надо хвост, а не всё подряд. Разовый замер отвечает на один вопрос, а регулярный покажет сдвиг после правок вроде DK-137 и вылезший новый жир. Кандидат: утилита рядом с regcheck, читает jsonl построчно, сводит вызовы по инструменту и по головной команде конвейера, объём вывода, перцентили одного вывода, повторные чтения и разбивку токенов из usage. Черновой скрипт приложен в файле черновика, привязка результата к вызову идёт по tool_use_id.

## Черновой скрипт

Тот самый разовый прогон, приложен целиком, чтобы замер повторялся до
оформления утилиты. Гоняется как `python3 tstats.py`, ходит по
`~/.claude/projects`, ничего не пишет.

```python
"""Сводка по транскриптам сессий: куда уходит вывод инструментов и токены."""

import json
import os
import re
import sys
from collections import Counter, defaultdict

ROOT = os.path.expanduser("~/.claude/projects")

# сколько вызовов, сколько символов вывода
calls = Counter()
out_chars = Counter()
# Bash: разбор по команде
bash_calls = Counter()
bash_chars = Counter()   # объём на головную команду вызова
bash_seen = Counter()    # объём вызовов, где команда участвовала хоть где-то
bash_head = Counter()    # вызовов, где команда головная
# распределение размеров вывода
sizes = []
sizes_by_tool = defaultdict(list)
# самые жирные одиночные выводы
fattest = []
# чтения файлов: повторы внутри одной сессии
repeat_by_file = Counter()
total_reads = 0
# токены
tok = Counter()
sessions = 0
lines_bad = 0
denied = Counter()

# sed бывает правкой файла и бывает чтением фрагмента, это разные вопросы
sed_kind = Counter()
sed_chars = Counter()
# читал ли Read кусок или файл целиком
read_kind = Counter()
read_chars = Counter()
# чтение файла средствами shell вместо Read
shell_read = Counter()
shell_read_chars = Counter()

SED_INPLACE = re.compile(r"\bsed\b[^|;&]*\s-[a-zA-Z]*i")
SED_PRINT = re.compile(r"\bsed\b[^|;&]*-n[^|;&]*[0-9,$]+\s*p")

MULTI = {"git", "go", "npm", "cargo", "python3", "python", "docker", "gh",
         "taskctl", "shipctl", "agentctl", "devkitctl", "make", "brew",
         "pnpm", "yarn", "kubectl", "systemctl", "pytest", "uv"}

SPLIT = re.compile(r"\|\||&&|\||;|\n")


# обломки многострочных скриптов и heredoc, командами не являются
NOISE = {"if", "fi", "then", "else", "elif", "for", "do", "done", "while",
         "case", "esac", "}", "{", "EOF", "PY", "SH", "import", "def", "class",
         "return", "in", "e", "f", "s", "print", "with", "try", "except",
         "from", "-", "--", "*", "*)", ")", "&&", "||", "exit", "break"}


def cmd_key(command):
    """Первое слово каждого сегмента конвейера, с уточнением для мультикоманд."""
    keys = []
    for seg in SPLIT.split(command or ""):
        parts = seg.strip().split()
        # пропускаем присваивания переменных и sudo
        while parts and ("=" in parts[0].split("/")[-1][:20] and not parts[0].startswith("-")
                         or parts[0] in ("sudo", "command", "time", "env")):
            parts = parts[1:]
        # «cd /path && cmd» это навигация, содержательная команда в следующем сегменте
        if not parts or parts[0] == "cd":
            continue
        head = os.path.basename(parts[0])
        if head in NOISE or not head or head.startswith(("\"", "'", "#", "$")):
            continue
        if head in MULTI and len(parts) > 1 and not parts[1].startswith("-"):
            head = head + " " + parts[1]
        keys.append(head)
    return keys


def walk():
    for dirpath, _, files in os.walk(ROOT):
        for name in files:
            if name.endswith(".jsonl"):
                yield os.path.join(dirpath, name)


for path in walk():
    sessions += 1
    pending = {}  # tool_use_id -> (tool, краткое описание)
    reads_here = Counter()  # повторы считаем внутри сессии, а не по всем
    try:
        fh = open(path, encoding="utf-8", errors="replace")
    except OSError:
        continue
    with fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            try:
                rec = json.loads(line)
            except (json.JSONDecodeError, ValueError):
                lines_bad += 1
                continue
            if not isinstance(rec, dict):
                continue
            msg = rec.get("message")
            if not isinstance(msg, dict):
                continue
            content = msg.get("content")

            if rec.get("type") == "assistant":
                u = msg.get("usage") or {}
                for k in ("input_tokens", "output_tokens",
                          "cache_creation_input_tokens", "cache_read_input_tokens"):
                    v = u.get(k)
                    if isinstance(v, int):
                        tok[k] += v
                if isinstance(content, list):
                    for blk in content:
                        if not isinstance(blk, dict) or blk.get("type") != "tool_use":
                            continue
                        tool = blk.get("name") or "?"
                        inp = blk.get("input") or {}
                        calls[tool] += 1
                        label = ""
                        extra = None
                        if tool == "Bash":
                            label = " ".join((inp.get("command") or "").split())[:120]
                            ks = cmd_key(inp.get("command"))
                            for k in set(ks):
                                bash_calls[k] += 1
                            if ks:
                                bash_head[ks[0]] += 1
                            cm = inp.get("command") or ""
                            if "sed" in set(ks):
                                if SED_INPLACE.search(cm):
                                    extra = ("sed", "правка файла (-i)")
                                elif SED_PRINT.search(cm):
                                    extra = ("sed", "чтение фрагмента (-n Np)")
                                else:
                                    extra = ("sed", "фильтр в конвейере")
                                sed_kind[extra[1]] += 1
                            elif ks and ks[0] in ("cat", "head", "tail"):
                                extra = ("shellread", ks[0])
                                shell_read[ks[0]] += 1
                        elif tool in ("Read", "Edit", "Write", "NotebookEdit"):
                            label = str(inp.get("file_path") or "")
                            if tool == "Read":
                                reads_here[label] += 1
                                if inp.get("limit") or inp.get("offset"):
                                    read_kind["кусок (limit/offset)"] += 1
                                elif inp.get("pages"):
                                    read_kind["страницы PDF"] += 1
                                else:
                                    read_kind["файл целиком"] += 1
                        elif tool in ("Grep", "Glob"):
                            label = str(inp.get("pattern") or "")
                        if tool == "Read":
                            extra = ("read", "кусок (limit/offset)"
                                     if (inp.get("limit") or inp.get("offset"))
                                     else "файл целиком")
                        pending[blk.get("id")] = (tool, label, inp.get("command"), extra)

            elif rec.get("type") == "user" and isinstance(content, list):
                for blk in content:
                    if not isinstance(blk, dict) or blk.get("type") != "tool_result":
                        continue
                    c = blk.get("content")
                    if isinstance(c, list):
                        text = "".join(
                            p.get("text", "") for p in c
                            if isinstance(p, dict) and p.get("type") == "text")
                    else:
                        text = c if isinstance(c, str) else ""
                    n = len(text)
                    key = blk.get("tool_use_id") or blk.get("id")
                    tool, label, raw, extra = pending.pop(key, ("?", "", None, None))
                    if extra:
                        kind, what = extra
                        if kind == "sed":
                            sed_chars[what] += n
                        elif kind == "read":
                            read_chars[what] += n
                        elif kind == "shellread":
                            shell_read_chars[what] += n
                    out_chars[tool] += n
                    sizes.append(n)
                    sizes_by_tool[tool].append(n)
                    if tool == "Bash" and raw:
                        keys = cmd_key(raw)
                        if keys:
                            # объём вызова вешаем на головную команду, иначе
                            # конвейер «git log | head» задваивает счёт
                            bash_chars[keys[0]] += n
                        for k in set(keys):
                            bash_seen[k] += n
                    low = text[:400].lower()
                    if blk.get("is_error") and (
                            "permission" in low or "не разреш" in low
                            or "requested permissions" in low or "user doesn't want" in low):
                        denied[tool] += 1
                    fattest.append((n, tool, label))
                    if len(fattest) > 4000:
                        fattest.sort(reverse=True)
                        del fattest[300:]
    for k, v in reads_here.items():
        total_reads += v
        if v > 1:
            repeat_by_file[k] += v - 1

fattest.sort(reverse=True)
del fattest[60:]


def pct(vals, p):
    if not vals:
        return 0
    s = sorted(vals)
    return s[min(len(s) - 1, int(len(s) * p / 100))]


def human(n):
    for unit in ("", "K", "M", "G"):
        if abs(n) < 1000:
            return f"{n:.0f}{unit}" if unit == "" else f"{n:.1f}{unit}"
        n /= 1000
    return f"{n:.1f}T"


total_out = sum(out_chars.values())
total_calls = sum(calls.values())

print(f"сессий: {sessions}, вызовов инструментов: {total_calls}, "
      f"битых строк: {lines_bad}")
print(f"суммарный вывод инструментов: {human(total_out)} символов "
      f"(~{human(total_out/3.5)} токенов)")
print()

print("=== ТОКЕНЫ ПО USAGE (все сессии) ===")
tot_tok = tok["input_tokens"] + tok["output_tokens"] + \
    tok["cache_creation_input_tokens"] + tok["cache_read_input_tokens"]
for k in ("cache_read_input_tokens", "cache_creation_input_tokens",
          "input_tokens", "output_tokens"):
    share = 100 * tok[k] / tot_tok if tot_tok else 0
    print(f"  {k:32} {human(tok[k]):>8}  {share:5.1f}%")
print()

print("=== ИНСТРУМЕНТЫ: вызовы и объём вывода ===")
print(f"{'инструмент':<14} {'вызовов':>8} {'%выз':>6} {'вывод':>8} {'%выв':>6} "
      f"{'медиана':>8} {'p90':>8} {'max':>9}")
for tool, n in calls.most_common(20):
    ch = out_chars[tool]
    ss = sizes_by_tool.get(tool, [])
    print(f"{tool:<14} {n:>8} {100*n/total_calls:>5.1f}% {human(ch):>8} "
          f"{100*ch/total_out if total_out else 0:>5.1f}% "
          f"{pct(ss,50):>8} {pct(ss,90):>8} {max(ss) if ss else 0:>9}")
print()

nbash = calls["Bash"]
print("=== BASH: топ команд по ЧИСЛУ вызовов (доля от вызовов Bash) ===")
for k, n in bash_calls.most_common(25):
    print(f"  {k:<26} в {n:>6} вызовах ({100*n/nbash:>4.1f}%), "
          f"головной в {bash_head[k]:>6}, вывод как головной {human(bash_chars[k]):>8}")
print()

bash_out = out_chars["Bash"]
print("=== BASH: топ команд по ОБЪЁМУ вывода (команда головная в вызове) ===")
for k, ch in bash_chars.most_common(25):
    n = bash_head[k]
    avg = ch / n if n else 0
    print(f"  {k:<26} {human(ch):>8} ({100*ch/bash_out if bash_out else 0:>4.1f}% Bash)  "
          f"вызовов {n:>5}  в среднем {human(avg):>7}")
print()

print("=== ЧЕМ БЫ ЭТО МОГЛО БЫТЬ: доля вызовов Bash с поиском и правкой текста ===")
for k in ("grep", "rg", "sed", "awk", "cat", "head", "tail", "find", "ls", "echo"):
    n = bash_calls.get(k, 0)
    print(f"  {k:<8} {n:>6} вызовов ({100*n/nbash if nbash else 0:>4.1f}%), "
          f"объём где головная: {human(bash_chars.get(k,0)):>8}")
print()

print("=== SED: зачем его зовут ===")
ts = sum(sed_kind.values())
for k, n in sed_kind.most_common():
    print(f"  {k:<28} {n:>6} ({100*n/ts if ts else 0:>4.1f}%)  вывод {human(sed_chars[k]):>8}")
print()

print("=== ЧТЕНИЕ ФАЙЛА: инструментом Read или средствами shell ===")
tr = sum(read_kind.values())
for k, n in read_kind.most_common():
    print(f"  Read, {k:<22} {n:>6} ({100*n/tr if tr else 0:>4.1f}%)  "
          f"вывод {human(read_chars[k]):>8}  в среднем {human(read_chars[k]/n if n else 0):>7}")
for k, n in shell_read.most_common():
    print(f"  shell, {k:<21} {n:>6}            вывод {human(shell_read_chars[k]):>8}  "
          f"в среднем {human(shell_read_chars[k]/n if n else 0):>7}")
print()

print("=== РАСПРЕДЕЛЕНИЕ РАЗМЕРА ОДНОГО ВЫВОДА ===")
for p in (50, 75, 90, 95, 99):
    print(f"  p{p:<3} {pct(sizes,p):>9} символов")
print(f"  max  {max(sizes) if sizes else 0:>9} символов")
big = [s for s in sizes if s > 20000]
print(f"  вызовов тяжелее 20K символов: {len(big)} "
      f"({100*len(big)/len(sizes) if sizes else 0:.2f}% вызовов, "
      f"{100*sum(big)/total_out if total_out else 0:.1f}% всего вывода)")
print()

print("=== ТОП-25 САМЫХ ЖИРНЫХ ОДИНОЧНЫХ ВЫВОДОВ ===")
for n, tool, label in fattest[:25]:
    print(f"  {human(n):>7}  {tool:<8} {label[:90]}")
print()

print("=== ПОВТОРНЫЕ ЧТЕНИЯ ОДНОГО ФАЙЛА ВНУТРИ СЕССИИ (топ 15) ===")
waste = sum(repeat_by_file.values())
print(f"  всего Read: {total_reads}, из них повторных внутри сессии: {waste} "
      f"({100*waste/total_reads if total_reads else 0:.1f}%)")
for k, v in repeat_by_file.most_common(15):
    print(f"  +{v:>4}  {k[:95]}")
print()

if denied:
    print("=== ОТКАЗЫ РАЗРЕШЕНИЙ ===")
    for k, v in denied.most_common(10):
        print(f"  {k:<14} {v}")
```
