```sh

[ -f "$f" ] || f=docs/tasks/OB-032.md

	{ echo "вопрос про внутренности задан виджетом: AskUserQuestion звался"; exit 1; }

	{ echo "внутренняя развилка «поиск» ушла человеку: решено человеком стоит в $f"; exit 1; }

	{ echo "внутренняя развилка «поиск» не переведена исполнителю: решает: исполнитель нет в $f"; exit 1; }
```
